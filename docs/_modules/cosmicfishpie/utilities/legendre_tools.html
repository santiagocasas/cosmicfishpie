def gauss_lobatto_abscissa_and_weights(order):
    lp = np.zeros(order, dtype="float")
    roots = np.zeros(order, dtype="float")
    lp[order - 1] = 1
    der = lg.legder(lp)
    roots[0] = 0
    roots[-1] = 1
    roots[1:-1] = lg.legroots(der)
    weights = 2.0 / (order * (order - 1) * lg.legval(roots, lp) ** 2)
    return roots, weights
